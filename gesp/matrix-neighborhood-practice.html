01低8 邻域 / 坐标标记

花坛旁的石头

返回题目

思路解析

先用二维数组把石头坐标标记为 1。枚举每个格子，如果它是石头就输出 #；否则用 8 个方向数组检查相邻位置是否在地图内，并累加石头数量。

参考代码（C++）

#include <iostream>
using namespace std;

int stone[505][505];
int dx[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
int dy[8] = {-1, 0, 1, -1, 1, -1, 0, 1};

int main() {
    int n, m, q;
    cin >> n >> m >> q;
    for (int i = 1; i <= q; i++) {
        int r, c;
        cin >> r >> c;
        stone[r][c] = 1;
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (stone[i][j] == 1) {
                cout << "#";
            } else {
                int cnt = 0;
                for (int d = 0; d < 8; d++) {
                    int x = i + dx[d];
                    int y = j + dy[d];
                    if (x >= 1 && x <= n && y >= 1 && y <= m && stone[x][y] == 1) {
                        cnt++;
                    }
                }
                cout << cnt;
            }
            if (j < m) cout << " ";
        }
        cout << "\n";
    }
    return 0;
}

02低4 邻域 / 字符矩阵

菜地四邻贫瘠度

返回题目

思路解析

这题只看上下左右，因此方向数组长度为 4。枚举普通土地时，逐个判断四邻格是否在地图内、是否为 P。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[205][205];
int dx[4] = {-1, 1, 0, 0};
int dy[4] = {0, 0, -1, 1};

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) {
            a[i][j] = s[j - 1];
        }
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == 'P' || a[i][j] == '#') {
                cout << a[i][j];
            } else {
                int cnt = 0;
                for (int d = 0; d < 4; d++) {
                    int x = i + dx[d];
                    int y = j + dy[d];
                    if (x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] == 'P') {
                        cnt++;
                    }
                }
                cout << cnt;
            }
            if (j < m) cout << " ";
        }
        cout << "\n";
    }
    return 0;
}

03低4 方向 / 位置检测

勇者朝向攻击

返回题目

思路解析

先根据方向把目标坐标算出来，再依次判断越界、墙、怪物、空地。注意越界要先判断，否则直接访问数组可能出错。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[105][105];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) {
            a[i][j] = s[j - 1];
        }
    }

    int q;
    cin >> q;
    for (int i = 1; i <= q; i++) {
        int r, c;
        char dir;
        cin >> r >> c >> dir;
        int nr = r, nc = c;
        if (dir == 'U') nr--;
        else if (dir == 'D') nr++;
        else if (dir == 'L') nc--;
        else nc++;

        if (nr < 1 || nr > n || nc < 1 || nc > m || a[nr][nc] == '#') {
            cout << "BLOCK\n";
        } else if (a[nr][nc] == 'M') {
            cout << "HIT\n";
        } else {
            cout << "MISS\n";
        }
    }
    return 0;
}

05中4 邻域 + 斜邻域

农场喷灌覆盖

返回题目

思路解析

枚举一个土地格周围 8 个位置。若相邻格和当前格同行或同列，就是 4 邻域；否则就是斜邻域。先判断 A，再判断 B。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[205][205];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) a[i][j] = s[j - 1];
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == 'S' || a[i][j] == '#') {
                cout << a[i][j];
            } else {
                int four = 0, diag = 0;
                for (int x = i - 1; x <= i + 1; x++) {
                    for (int y = j - 1; y <= j + 1; y++) {
                        if (x == i && y == j) continue;
                        if (x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] == 'S') {
                            if (x == i || y == j) four++;
                            else diag++;
                        }
                    }
                }
                if (four > 0) cout << "A";
                else if (diag > 0) cout << "B";
                else cout << "0";
            }
        }
        cout << "\n";
    }
    return 0;
}

06中4 邻域条件 + 8 邻域排除

海岛救援站选址

返回题目

思路解析

这题有两个条件：4 邻域找村庄，8 邻域排除海水。两个条件都满足时计数加一并输出 R。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[205][205];
int dx4[4] = {-1, 1, 0, 0};
int dy4[4] = {0, 0, -1, 1};
int dx8[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
int dy8[8] = {-1, 0, 1, -1, 1, -1, 0, 1};

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) a[i][j] = s[j - 1];
    }

    char ans[205][205];
    int total = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            ans[i][j] = a[i][j];
            if (a[i][j] != '.') continue;

            int hasCity = 0, hasSea = 0;
            for (int d = 0; d < 4; d++) {
                int x = i + dx4[d], y = j + dy4[d];
                if (x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] == 'C') hasCity = 1;
            }
            for (int d = 0; d < 8; d++) {
                int x = i + dx8[d], y = j + dy8[d];
                if (x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] == '~') hasSea = 1;
            }
            if (hasCity == 1 && hasSea == 0) {
                ans[i][j] = 'R';
                total++;
            }
        }
    }

    cout << total << "\n";
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) cout << ans[i][j];
        cout << "\n";
    }
    return 0;
}

枚举每个弓箭塔，再沿 4 个方向一步步走，直到越界或遇到墙。走到空地时让该格警戒次数加一。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[205][205];
int cnt[205][205];
int dx[4] = {-1, 1, 0, 0};
int dy[4] = {0, 0, -1, 1};

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) a[i][j] = s[j - 1];
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == 'A') {

    }

08高8 方向 / 距离 2 / 障碍阻挡

魔法爆炸安全区

返回题目

思路解析

对每个炸弹枚举 8 个方向，每个方向最多走 2 步。若越界或遇到墙就停止；若走到空地则危险次数加一。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[205][205];
int cnt[205][205];
int dx[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
int dy[8] = {-1, 0, 1, -1, 1, -1, 0, 1};

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) a[i][j] = s[j - 1];
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == 'B') {
                for (int d = 0; d < 8; d++) {
                    for (int step = 1; step <= 2; step++) {
                        int x = i + dx[d] * step;
                        int y = j + dy[d] * step;
                        if (x < 1 || x > n || y < 1 || y > m) break;
                        if (a[x][y] == '#') break;
                        if (a[x][y] == '.') cnt[x][y]++;
                    }
                }
            }
        }
    }

    int safe = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == '.' && cnt[i][j] == 0) safe++;
        }
    }
    cout << safe << "\n";

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == '.') cout << cnt[i][j];
            else cout << a[i][j];
            if (j < m) cout << " ";
        }
        cout << "\n";
    }
    return 0;
}

09高方向字符 / 直线检测

机器人单向巡逻

返回题目

思路解析

每个机器人只有一个方向。根据方向字符得到 dx、dy，然后沿直线前进，遇到 # 或越界停止；空地计数加一。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[205][205];
int cnt[205][205];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) a[i][j] = s[j - 1];
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            int dx = 0, dy = 0;
            if (a[i][j] == '^') dx = -1;
            else if (a[i][j] == 'v') dx = 1;
            else if (a[i][j] == '<') dy = -1;
            else if (a[i][j] == '>') dy = 1;
            else continue;

            int x = i + dx, y = j + dy;
            while (x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] != '#') {
                if (a[x][y] == '.') cnt[x][y]++;
                x += dx;
                y += dy;
            }
        }
    }

    int total = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == '.' && cnt[i][j] > 0) total++;
        }
    }
    cout << total << "\n";

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] == '.') cout << cnt[i][j];
            else cout << a[i][j];
            if (j < m) cout << " ";
        }
        cout << "\n";
    }
    return 0;
}

10高8 邻域计数 + 4 邻域通路

防疫隔离点标记

返回题目

思路解析

对每个空地分别统计 8 邻域感染点数量和 4 邻域空地数量。两个条件同时满足才标记，原来的 I 和 # 不改变。

参考代码（C++）

#include <iostream>
#include <string>
using namespace std;

char a[205][205], ans[205][205];
int dx4[4] = {-1, 1, 0, 0};
int dy4[4] = {0, 0, -1, 1};
int dx8[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
int dy8[8] = {-1, 0, 1, -1, 1, -1, 0, 1};

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for (int j = 1; j <= m; j++) {
            a[i][j] = s[j - 1];
            ans[i][j] = a[i][j];
        }
    }

    int total = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (a[i][j] != '.') continue;
            int infected = 0, exits = 0;
            for (int d = 0; d < 8; d++) {
                int x = i + dx8[d], y = j + dy8[d];
                if (x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] == 'I') infected++;
            }
            for (int d = 0; d < 4; d++) {
                int x = i + dx4[d], y = j + dy4[d];
                if (x >= 1 && x <= n && y >= 1 && y <= m && a[x][y] == '.') exits++;
            }
            if (infected >= 2 && exits >= 1) {
                ans[i][j] = '!';
                total++;
            }
        }
    }

    cout << total << "\n";
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) cout << ans[i][j];
        cout << "\n";
    }
    return 0;
}
